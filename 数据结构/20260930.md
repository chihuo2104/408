## 利用栈来解决迷宫问题

给定一个M * N的迷宫图，求一条从指定入口到出口的迷宫路径，在行走中每一步只能从当前方块移动到上、下、左、右相邻方块中的一个方块。假设迷宫如下图所示，其中每个方块用空白表示通道，用阴影表示障碍物。

为了表示迷宫，设置一个数组mg，其中每个元素表示一个方块的状态，为0时表示对应方块时通道，为1时表示对应方块是障碍物。为了算法方便，一般在迷宫的外围加上一条围墙。则上图迷宫数组表示如下：


### 求解思路

对于迷宫中的每个方块，有上、下、左、右4个方块相邻，第i行第j列的当前方块位置记为(i,j)，规定上方方块为方位0，并按顺时针方向递增编号（如下图）。在试探过程中，统一从方位0到方位3的方向查找下一个可走的相邻方块。

为了保证在任何位置都能沿原路退回（回溯），需要保存从入口到当前位置的路径上走过的方块，由于回溯的过程是从当前位置退到前一个方块，体现出后进先出的特点，所以采用栈保存走过的方块。

```c
bool mgpath(int xi, int yi, int xe, int ye)
//搜索(xi,yi)到(xe,ye)的一条路径
{
    Box path[MaxSize], e;
    int i, j, di, i1, j1, k;
    bool find;
    StType *st;
    InitStack(st);
    e.i = xi;
    e.j = yi;
    e.di = -1;
    Push(st, e);
    mg[xi][yi] = -1;

    while (!StackEmpty(st))
    {
        GetTop(st, e);
        i = e.i;
        j = e.j;
        di = e.di;

        if (i == xe && j == ye)
        {
            printf("一条迷宫路径如下:\n");
            k = 0;
            while (!StackEmpty(st))
            {
                Pop(st, e);
                path[k++] = e;
            }
            while (k > 0)
            {
                printf("\t(%d,%d)", path[k - 1].i, path[k - 1].j);
                if ((k + 1) % 5 == 0)
                    printf("\n");
                k--;
            }
            printf("\n");
            DestroyStack(st);
            return true;
        }

        find = false;
        while (di < 4 && !find)   // 找方块(i,j)的下一个相邻可走方块(i1,j1)
        {
            /**********begin*********/
            di++;
            if (di > 3) break;    // 四个方向都已试探完
            switch (di)
            {
                case 0: i1 = i - 1; j1 = j;     break;  // 上
                case 1: i1 = i;     j1 = j + 1; break;  // 右
                case 2: i1 = i + 1; j1 = j;     break;  // 下
                case 3: i1 = i;     j1 = j - 1; break;  // 左
            }
            /**********end*********/
            if (mg[i1][j1] == 0) find = true;
        }

        if (find)  // 找到相邻可走方块则入栈操作
        {
            /**********begin*********/
            Pop(st, e);          // 弹出当前栈顶
            e.di = di;           // 记录当前已经试探到的方向
            Push(st, e);         // 重新压入，更新其 di 值

            e.i = i1;            // 新方块入栈
            e.j = j1;
            e.di = -1;
            Push(st, e);

            mg[i1][j1] = -1;     // 标记为已走过
            /**********end*********/
        }
        else  // 没有路径可走，则退栈
        {
            /**********begin*********/
            Pop(st, e);          // 回溯：弹出当前方块
            mg[e.i][e.j] = -1;   // 保持已访问标记，避免重复走
            /**********end*********/
        }
    }

    DestroyStack(st);
    return false;
}
```

## 利用队列来求解迷宫问题

给定一个M * N的迷宫图，求一条从指定入口到出口的迷宫路径，在行走中每一步只能从当前方块移动到上、下、左、右相邻方块中的一个方块。假设迷宫如下图所示(图略），其中每个方块用空白表示通道，用阴影表示障碍物。

用队列解决求解迷宫问题，借助一个顺序队qu保存走过的方块，其声明如下：

```c
typedef struct
{
    int i,j;
    int pre;
} Box;

typedef struct
{
    Box data[MaxSize];
    int front,rear;
} QuType;
```

设计思路

搜索从入口 `(xi,yi)` 到出口 `(xe,ye)` 路径的过程是，首先将入口 `(xi,yi)` 进队，在队列 qu 不为空时循环，出队一个方块 e，然后查找方块 e 的所有相邻可走方块，假设为 e1 和 e2 两个方块，将它们进队，它们在队列中的位置用 rear1 和 rear2 表示，并且将它们的 pre 设置为 front，如图所示。（图略）

一条迷宫的路径如下:

```
(1,1)   (2,1)   (3,1)   (4,1)   (5,1)
(5,2)   (5,3)   (6,3)   (6,4)   (6,5)
(7,5)   (8,5)   (8,6)   (8,7)   (8,8)
```

```c
{
```

```c
		/*********begin********/
		path[k]=qu->data[p];
		k++;
		p=qu->data[p].pre;
		/*********end********/
	}

bool mgpath1(int xi,int yi, int xe, int ye)
//查找(xi,yi)到(xe,ye)的搜索路径
{
	Box e;
	int i,j,di,i1,j1;
	QuType *qu;
	InitQueue(qu);

	e.i=xi;
	e.j=yi;
	e.pre=-1;
	enQueue(qu,e);

	mg[xi][yi]=-1;

	while(!QueueEmpty(qu))
	{
		deQueue(qu,e);
		i=e.i;
		j=e.j;

		if(i==xe && j==ye) //找到出口，输出路径
		{
			/*********begin********/
			dispapath(qu,qu->front);
			DestroyQueue(qu);
			return true;
			/*********end********/
		}

		for(di=0;di<4;di++) //循环遍历每个方位
		{
			/*********begin********/
			switch(di)
			{
				case 0:
					i1=i-1;
					j1=j;
					break;
				case 1:
					i1=i;
					j1=j+1;
					break;
				case 2:
					i1=i+1;
					j1=j;
					break;
				case 3:
					i1=i;
					j1=j-1;
					break;
			}

			if(mg[i1][j1]==0) //该方块可走
			{
				e.i=i1;
				e.j=j1;
				e.pre=qu->front;
				enQueue(qu,e);
				mg[i1][j1]=-1; //标记已经访问
			}
			/*********end********/
		}
	}

	DestroyQueue(qu);
	return false;
```

## 约瑟夫环问题

约瑟夫问题是个著名的问题：n 个人围成一圈 (3<=n<=100)（从零开始编号），从第一个人开始报数，第 m 个将出列(1<=m<=n)，重复报数，直到所有人都出列 。例如n = 6，m = 5，出列的顺序为：5, 4, 6, 2, 3, 1。

单链表节点类型定义如下：

```c
typedef struct LNode{
    ElemType data;
    struct LNode *next;
};
```

### 编程要求

设计一个算法利用循环单链表解决约瑟夫环问题。
程序包含两个输入：n和m
测试说明

平台会对你编写的代码进行测试：

### 测试输入：

```
6 5
```

### 预期输出：

```
5 4 6 2 3 1
```

*n 个人围成一圈 (3<=n<=100)（从1开始编号），从第一个人开始报数，第 m 个人出列 (1<=m<=n)，需要给出所有人出列的顺序

### 参考答案（DeepSeek V4.1 Flash）

```c
#include <stdio.h>
#include <stdlib.h>

typedef int ElemType;

typedef struct LNode {
    ElemType data;
    struct LNode *next;
} LNode, *LinkList;

void Josephus(int n, int m) {
    if (n <= 0 || m <= 0) return;

    // 创建第一个结点
    LinkList head = (LinkList)malloc(sizeof(LNode));
    head->data = 1;
    head->next = NULL;
    LinkList tail = head;

    // 尾插法创建剩余结点 2 ~ n
    for (int i = 2; i <= n; i++) {
        LinkList p = (LinkList)malloc(sizeof(LNode));
        p->data = i;
        p->next = NULL;
        tail->next = p;
        tail = p;
    }

    // 构成循环单链表
    tail->next = head;

    // prev 指向第一个结点的前驱，即最后一个结点
    LinkList prev = tail;

    for (int i = 0; i < n; i++) {
        // 从 prev->next 开始报数，报 m 的人出列
        // 所以 prev 向后移动 m-1 次
        for (int j = 1; j < m; j++) {
            prev = prev->next;
        }

        LinkList q = prev->next;  // q 为要出列的结点
        if (i > 0) printf(" ");
        printf("%d", q->data);

        // 删除 q
        prev->next = q->next;
        free(q);
    }

    printf("\n");
}

int main() {
    int n, m;
    scanf("%d %d", &n, &m);
    Josephus(n, m);
    return 0;
}
```
