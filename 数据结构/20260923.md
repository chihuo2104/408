## 环形队列基本运算

### 任务描述

实现环形队列的各种基本运算。

### 测试输入：

```
abc def
```

### 预期输出：

```
空队！
a
b c d e f
```

### 参考答案

```c
//顺序队列（环形队列）基本运算算法
#include <stdio.h>
#include <malloc.h>
#define MaxSize 100

typedef char ElemType;

typedef struct 
{	
    ElemType data[MaxSize];
    int front, rear;        //队首和队尾指针
} SqQueue;

void InitQueue(SqQueue *&q) //初始化队列
{		
    /************begin***********/
    q = (SqQueue *)malloc(sizeof(SqQueue));
    q->front = q->rear = 0;
    /************end***********/
}

void DestroyQueue(SqQueue *&q)
{
    free(q);
}

bool QueueEmpty(SqQueue *q) //判断队是否为空
{
    /************begin***********/
    return (q->front == q->rear);
    /************end***********/
}

bool enQueue(SqQueue *&q, ElemType e) //入队
{
    if ((q->rear + 1) % MaxSize == q->front) //队满上溢出
        return false;

    /************begin***********/
    q->data[q->rear] = e;
    q->rear = (q->rear + 1) % MaxSize;
    /************end***********/

    return true;
}

bool deQueue(SqQueue *&q, ElemType &e) //出队
{
    if (q->front == q->rear) //队空下溢出
        return false;

    /************begin***********/
    e = q->data[q->front];
    q->front = (q->front + 1) % MaxSize;
    /************end***********/

    return true;
}
```

## 链队基本运算

### 任务描述

实现链队的各种基本运算。

### 测试输入：

```
abcdef
```

### 预期输出：

```
空队！
a

```

### 参考答案

```c
//链队运算算法
#include <stdio.h>
#include <malloc.h>
typedef char ElemType;

typedef struct DataNode
{	
	ElemType data;
	struct DataNode *next;
} DataNode;				//链队数据结点类型

typedef struct
{	
	DataNode *front;
	DataNode *rear;
} LinkQuNode;			//链队类型

void InitQueue(LinkQuNode *&q)//初始化队列
{	
	/************begin************/
	q=(LinkQuNode *)malloc(sizeof(LinkQuNode));
	q->front=q->rear=NULL;
    /**************end*************/
}

void DestroyQueue(LinkQuNode *&q)
{
	DataNode *p=q->front,*r;//p指向队头数据结点
	if (p!=NULL)			//释放数据结点占用空间
	{	
		r=p->next;
		while (r!=NULL)
		{	
			free(p);
			p=r;
			r=p->next;
		}
	}
	free(p);
	free(q);				//释放链队结点占用空间
}

bool QueueEmpty(LinkQuNode *q)//判断队列是否为空
{
	/************begin************/
	return(q->rear==NULL);
    /**************end*************/
}

void enQueue(LinkQuNode *&q,ElemType e)//入队
{	
    /************begin************/
	DataNode *p;
	p=(DataNode *)malloc(sizeof(DataNode));
	p->data=e;
	p->next=NULL;

	if (q->rear==NULL)		//原队列为空
		q->front=q->rear=p;
	else
	{
		q->rear->next=p;
		q->rear=p;
	}
    /**************end*************/
}

bool deQueue(LinkQuNode *&q,ElemType &e)//出队
{	
	DataNode *t;
	if (q->rear==NULL)		//队列为空
		return false;

	/************begin************/
	t=q->front;
	e=t->data;
	q->front=q->front->next;

	if (q->front==NULL)		//原来只有一个结点
		q->rear=NULL;

	free(t);
    /**************end*************/

	return true;
}
```

## 链栈基本操作

实现链栈各种基本运算。

### 参考答案

```c
//链栈基本运算算法

#include <stdio.h>
#include <malloc.h>

typedef char ElemType;

typedef struct linknode
{   
    ElemType data;              //数据域
    struct linknode *next;      //指针域
} LinkStNode;                   //链栈类型

void InitStack(LinkStNode *&s)
//初始化一个空栈
{
    s=(LinkStNode *)malloc(sizeof(LinkStNode));
    s->next=NULL;
}

void DestroyStack(LinkStNode *&s)
{
    LinkStNode *p=s->next;

    while (p!=NULL)
    {   
        free(s);
        s=p;
        p=p->next;
    }

    free(s);    //s指向尾结点,释放其空间
}

bool StackEmpty(LinkStNode *s)
//判断栈s是否为空，若是返回true，否则返回false
{
    return(s->next==NULL);
}

void Push(LinkStNode *&s,ElemType e)
//将元素e入栈s
{   
    LinkStNode *p;

    p=(LinkStNode *)malloc(sizeof(LinkStNode));
    p->data=e;

    p->next=s->next;
    s->next=p;
}

bool Pop(LinkStNode *&s,ElemType &e)
//将栈顶元素出栈
{   
    LinkStNode *p;

    if (s->next==NULL)
        return false;

    p=s->next;
    e=p->data;

    s->next=p->next;
    free(p);

    return true;
}

bool GetTop(LinkStNode *s,ElemType &e)
//获得栈顶元素
{   
    if (s->next==NULL)
        return false;

    e=s->next->data;

    return true;
}
```
