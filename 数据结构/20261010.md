## 求一个矩阵的马鞍点

### 任务描述

一个给定的二维矩阵中查找是否存在“马鞍点”。

马鞍点是指矩阵中的一个元素，它在所在行中是最小值，同时在所在列中是最大值。如果存在马鞍点，输出该马鞍点的值及其位置；如果不存在马鞍点，输出相应的信息。

根据提示，在右侧编辑器补充代码，寻找矩阵的马鞍点。

### 输入说明

第一行包含两个整数 n 和 m，分别表示矩阵的行数和列数。

接下来的 n 行，每行包含 m 个整数，表示矩阵的元素。

### 输出说明

如果存在马鞍点，输出马鞍点的值及其位置，格式为 `A[i][j]=value` ，其中 `i` 和 `j` 分别是马鞍点的行索引和列索引（从0开始计数）。

如果不存在马鞍点，输出 没有马鞍点。

### 测试输入输出

测试输入1：

```
3 3
1 2 3
2 3 1
3 1 2
```

预期输出1：

```
没有马鞍点
```

测试输入2：

```
3 3
1 2 3
4 5 6
7 8 9
```

预期输出2：

```
A[2][0]=7
```

```c
for (i = 0; i < n; i++) {
    // 找到第i行的最小值
    int min = matrix[i][0];

    for (j = 1; j < m; j++) {
        if (matrix[i][j] < min) {
            min = matrix[i][j];
        }
    }

    // 检查第i行中所有等于最小值的元素
    for (j = 0; j < m; j++) {
        if (matrix[i][j] == min) {
            bool isMax = true;

            // 判断该元素是否为所在列的最大值
            for (int k = 0; k < n; k++) {
                if (matrix[k][j] > matrix[i][j]) {
                    isMax = false;
                    break;
                }
            }

            // 找到马鞍点
            if (isMax) {
                printf("A[%d][%d]=%d\n", i, j, matrix[i][j]);
                found = true;
                break;
            }
        }
    }

    if (found) {
        break;
    }
}

// 如果没有找到马鞍点
if (!found) {
    printf("没有马鞍点\n");
}
```

### 参考答案

```c

#include <stdio.h>
#define MaxSize 100

typedef int ElemType;

// 定义三元组
typedef struct
{
    int r;
    int c;
    ElemType d;
} TupNode;

// 定义稀疏矩阵
typedef struct
{
    int rows;
    int cols;
    int nums;
    TupNode data[MaxSize * MaxSize];
} TSMatrix;

// 创建三元组
void CreateMat(TSMatrix &t, ElemType (*A)[MaxSize], int M, int N)
{
    int i, j;
    t.rows = M;
    t.cols = N;
    t.nums = 0;

    for (i = 0; i < M; i++)
    {
        for (j = 0; j < N; j++)
        {
            if (A[i][j] != 0)
            {
                t.data[t.nums].r = i;
                t.data[t.nums].c = j;          // 第1空
                t.data[t.nums].d = A[i][j];
                t.nums = t.nums + 1;           // 第2空
            }
        }
    }
}

// 给指定位置的元素赋值
bool Value(TSMatrix &t, ElemType x, int i, int j)
{
    int k = 0, k1;

    if (i < 0 || j < 0 || i >= t.rows || j >= t.cols)
        return false;

    while (k < t.nums && i > t.data[k].r)
        k = k + 1;                           // 第3空

    while (k < t.nums && i == t.data[k].r &&
           j > t.data[k].c)                  // 第4空
        k++;

    if (k < t.nums && t.data[k].r == i &&
        t.data[k].c == j)
    {
        if (x == 0)
        {
            // 赋值为0时删除该三元组
            for (k1 = k; k1 < t.nums - 1; k1++)
                t.data[k1] = t.data[k1 + 1];
            t.nums--;
        }
        else
        {
            t.data[k].d = x;                // 第5空
        }
    }
    else if (x != 0)
    {
        // 插入新的非零元素
        for (k1 = t.nums - 1; k1 >= k; k1--)
        {
            t.data[k1 + 1].r = t.data[k1].r;
            t.data[k1 + 1].c = t.data[k1].c;
            t.data[k1 + 1].d = t.data[k1].d;
        }

        t.data[k].r = i;
        t.data[k].c = j;
        t.data[k].d = x;
        t.nums++;
    }

    return true;
}

// 查询指定位置的元素
bool Assign(TSMatrix t, ElemType &x, int i, int j)
{
    int k = 0;

    if (i < 0 || j < 0 || i >= t.rows || j >= t.cols)
        return false;

    while (k < t.nums && i > t.data[k].r)
        k++;

    while (k < t.nums && i == t.data[k].r &&
           j > t.data[k].c)
        k++;

    if (k < t.nums && t.data[k].r == i &&
        t.data[k].c == j)
    {
        x = t.data[k].d;
    }
    else
    {
        x = 0;                              // 第6空
    }

    return true;
}

// 输出三元组
void DispMat(TSMatrix t)
{
    int k;

    if (t.nums <= 0)
        return;

    printf("\t%d\t%d\t%d\n", t.rows, t.cols, t.nums);
    printf("\t----------------------------------------\n");

    for (k = 0; k < t.nums; k++)            // 第7空
    {
        printf("\t%d\t%d\t%d\n",
               t.data[k].r,
               t.data[k].c,
               t.data[k].d);
    }
}

int main()
{
    int i, j, m, n;
    int px, py;

    scanf("%d %d", &m, &n);

    ElemType A[MaxSize][MaxSize] = {0};

    // 输入矩阵元素
    for (i = 0; i < m; i++)
    {
        for (j = 0; j < n; j++)
        {
            scanf("%d", &A[i][j]);
        }
    }

    ElemType x;
    TSMatrix t;

    // 创建并输出三元组
    CreateMat(t, A, m, n);
    DispMat(t);

    // 输入要查询的位置
    scanf("%d %d", &px, &py);

    // 查询元素
    Assign(t, x, px, py);
    printf("\tA[%d][%d]=%d\n", px, py, x);

    // 输入新值
    scanf("%d", &x);

    // 修改元素
    Value(t, x, px, py);

    // 输出修改后的三元组
    DispMat(t);

    return 0;
}
```
